local function u613(target)

function SuperPredictor.NormalizeDir(moveDir, axis, velocity)
    if axis == 'X' then
        if moveDir.X == 0 then
            return velocity.Unit.X ~= velocity.Unit.X and 1 or velocity.Unit.X
        end
        return moveDir.X
    elseif axis == 'Z' then
        if moveDir.Z == 0 then
            return velocity.Unit.Z ~= velocity.Unit.Z and 1 or velocity.Unit.Z
        end
        return moveDir.Z
    end
    return 0
end

function SuperPredictor.IsAxisAligned(part)
    if not part then return false end
    local orient = part.Orientation
    if math.abs(orient.X) == 0 or math.abs(orient.X) == 180 then
        if math.abs(orient.Z) == 0 or math.abs(orient.Z) == 180 then
            return true
        end
    end
    return false
end

function SuperPredictor.DetectFloor(part, direction, position, hrp)
    if not part or not hrp then return nil end
    
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {part}
    params.FilterType = Enum.RaycastFilterType.Include
    
    local dir = direction == 1 and -2 or 1
    local result = workspace:Blockcast(
        CFrame.new(Vector3.new(position.X, position.Y + dir / math.abs(dir), position.Z)) * CFrame.Angles(0, math.rad(hrp.Orientation.Y), 0),
        Vector3.new(2, 0.002, 1),
        Vector3.new(0, dir, 0),
        params
    )
    
    if result and result.Position then
        return result.Position.Y
    end
    return nil
end

function SuperPredictor.RevertPrediction(target, hrp, humanoid, pos, vel, unit, moveDir)
    local ping = SuperPredictor.Config.PredictionPing or 100
    local verticalDiv = (SuperPredictor.Config.RevertVerticalDivider or 100) / 100
    local horizontalDiv = (SuperPredictor.Config.RevertHorizontalDivider or 100) / 100
    local simDiv = -((SuperPredictor.Config.RevertSimulationDivider or 20) / 100)
    local interval = 1 / (SuperPredictor.Config.RevertPredictionInterval or 70)
    local predictJump = SuperPredictor.Config.PredictJump
    
    local startTime = tick()
    local orientationY = hrp.Orientation.Y
    local position = pos
    local velocity = vel
    local walkToPoint = humanoid.WalkToPoint
    local moveDir2 = moveDir
    
    if moveDir2 == Vector3.new() and walkToPoint ~= Vector3.new() then
        moveDir2 = CFrame.new(position * Vector3.new(1, 0, 1), walkToPoint * Vector3.new(1, 0, 1)).LookVector
    end
    
    -- Part simulasi
    local simParts = {
        Instance.new('Part'),
        Instance.new('Part'),
        Instance.new('Part'),
    }
    
    for _, part in pairs(simParts) do
        part.Transparency = 1
        part.Anchored = true
        part.Parent = workspace
    end
    
    simParts[1].Size = Vector3.new(2, 2, 1)
    simParts[2].Size = Vector3.new(2, 2, 1)
    simParts[3].Size = Vector3.new(2, 1, 1)
    
    simParts[1].CFrame = CFrame.new(position - Vector3.new(0, 2, 0)) * CFrame.Angles(0, math.rad(orientationY), 0)
    simParts[2].CFrame = CFrame.new(position) * CFrame.Angles(0, math.rad(orientationY), 0)
    simParts[3].CFrame = CFrame.new(position + Vector3.new(0, 1.5, 0)) * CFrame.Angles(0, math.rad(orientationY), 0)
    
    local blacklist = {simParts[1], simParts[2], simParts[3], target.Character, u135()}
    local rayParams = RaycastParams.new()
    rayParams.FilterDescendantsInstances = blacklist
    rayParams.FilterType = Enum.RaycastFilterType.Blacklist
    
    local elapsed = 0
    local step = interval
    local speed = 750
    
    while true do
        local timeElapsed = tick() - startTime
        local predictedTime = ping / 1100 + timeElapsed
        
        if predictedTime - elapsed > step then
            step = interval
        else
            step = predictedTime - elapsed or interval
        end
        
        local currentPos = position + Vector3.new(0, simDiv, 0)
        local nextPos = currentPos - Vector3.new(0, simDiv, 0)
        
        if predictedTime <= elapsed or 60 <= elapsed or nextPos.Y <= workspace.FallenPartsDestroyHeight then
            break
        end
        
        if walkToPoint ~= Vector3.new() then
            moveDir2 = CFrame.new(hrp.Position * Vector3.new(1, 0, 1), walkToPoint * Vector3.new(1, 0, 1)).LookVector or moveDir2
        end
        
        position = nextPos + Vector3.new(
            (velocity.X * step + speed * moveDir2.X * step^2 / 2) * horizontalDiv,
            (velocity.Y * step + -196.2 * step^2 / 2) * verticalDiv,
            (velocity.Z * step + speed * moveDir2.Z * step^2 / 2) * horizontalDiv
        )
        
        -- Update velocity
        for _, axis in ipairs({'X', 'Z'}) do
            local targetVel = moveDir2[axis] * 16
            local currentVel = velocity[axis]
            
            if targetVel >= currentVel then
                if currentVel >= targetVel then
                    if axis == 'X' then
                        velocity = Vector3.new(targetVel, velocity.Y, velocity.Z)
                    else
                        velocity = Vector3.new(velocity.X, velocity.Y, targetVel)
                    end
                else
                    local accel = speed * math.abs(SuperPredictor.NormalizeDir(moveDir2, axis, velocity)) * step
                    if axis == 'X' then
                        velocity = velocity + Vector3.new(accel, 0, 0)
                    else
                        velocity = velocity + Vector3.new(0, 0, accel)
                    end
                    if targetVel < velocity[axis] then
                        if axis == 'X' then
                            velocity = Vector3.new(targetVel, velocity.Y, velocity.Z)
                        else
                            velocity = Vector3.new(velocity.X, velocity.Y, targetVel)
                        end
                    end
                end
            else
                local decel = speed * math.abs(SuperPredictor.NormalizeDir(moveDir2, axis, velocity)) * step
                if axis == 'X' then
                    velocity = velocity - Vector3.new(decel, 0, 0)
                else
                    velocity = velocity - Vector3.new(0, 0, decel)
                end
                if velocity[axis] < targetVel then
                    if axis == 'X' then
                        velocity = Vector3.new(targetVel, velocity.Y, velocity.Z)
                    else
                        velocity = Vector3.new(velocity.X, velocity.Y, targetVel)
                    end
                end
            end
        end
        
        velocity = velocity - Vector3.new(0, 196.2 * step, 0)
        
        -- Deteksi tabrakan
        local bottomCheck = workspace:Blockcast(
            CFrame.new(currentPos - Vector3.new(0, 1.001, 0)) * CFrame.Angles(0, math.rad(orientationY), 0),
            Vector3.new(2, 0.002, 1),
            currentPos - Vector3.new(0, 1.001, 0) - (position - Vector3.new(0, 1.001, 0)),
            rayParams
        )
        
        local topCheck = workspace:Blockcast(
            CFrame.new(currentPos + Vector3.new(0, 1.001, 0)) * CFrame.Angles(0, math.rad(orientationY), 0),
            Vector3.new(2, 0.002, 1),
            currentPos + Vector3.new(0, 1.001, 0) - (position + Vector3.new(0, 1.001, 0)),
            rayParams
        )
        
        if bottomCheck and bottomCheck.Position and bottomCheck.Position.Y <= currentPos.Y then
            position = Vector3.new(position.X, bottomCheck.Position.Y + 3, position.Z)
        elseif topCheck and topCheck.Position and topCheck.Position.Y >= currentPos.Y then
            position = Vector3.new(position.X, topCheck.Position.Y - 2, position.Z)
        end
        
        simParts[1].Position = position - Vector3.new(0, 2, 0)
        simParts[2].Position = position
        simParts[3].Position = position + Vector3.new(0, 1.5, 0)
        
        -- Deteksi lantai
        local touchingBottom = simParts[2]:GetTouchingParts()
        local hitGround = false
        
        for _, part in ipairs(touchingBottom) do
            if part.ClassName == 'TrussPart' then
                hitGround = true
            end
        end
        
        if not hitGround and #touchingBottom > 0 and velocity.Y <= step then
            local highestY = nil
            
            for _, part in ipairs(touchingBottom) do
                local partPos = part.Position
                local partSize = part.Size
                
                if part.ClassName ~= 'Part' or part.Shape ~= Enum.PartType.Block or not SuperPredictor.IsAxisAligned(part) or (position.Y - 1 <= partPos.Y + partSize.Y / 2) or (highestY and highestY >= partPos.Y + partSize.Y / 2) then
                    local detectedY = SuperPredictor.DetectFloor(part, 1, position, hrp)
                    if detectedY then
                        if not highestY or highestY < detectedY then
                            highestY = detectedY
                        end
                    end
                else
                    highestY = partPos.Y + partSize.Y / 2
                end
            end
            
            if highestY then
                speed = 750
                position = Vector3.new(position.X, highestY + 3, position.Z)
                velocity = velocity * Vector3.new(1, 0, 1)
                
                if predictJump and humanoid.Jump then
                    velocity = Vector3.new(velocity.X, humanoid.JumpPower, velocity.Z)
                end
            else
                speed = 142
            end
        end
        
        simParts[1].Position = position - Vector3.new(0, 2, 0)
        simParts[2].Position = position
        simParts[3].Position = position + Vector3.new(0, 1.5, 0)
        
        -- Deteksi atap
        local touchingTop = simParts[3]:GetTouchingParts()
        local hitCeiling = false
        
        if not hitCeiling and #touchingTop > 0 and velocity.Y >= -step then
            local lowestY = nil
            
            for _, part in ipairs(touchingTop) do
                local partPos = part.Position
                local partSize = part.Size
                
                if (part.ClassName == 'Part' or part.ClassName == 'SpawnLocation') and (part.Shape == Enum.PartType.Block and SuperPredictor.IsAxisAligned(part)) and (position.Y + 1.9 < partPos.Y - partSize.Y / 2 and (not lowestY or partPos.Y - partSize.Y / 2 < lowestY)) then
                    lowestY = partPos.Y - partSize.Y / 2
                else
                    local detectedY = SuperPredictor.DetectFloor(part, 2, position, hrp)
                    if detectedY then
                        if not lowestY or detectedY < lowestY then
                            lowestY = detectedY
                        end
                    end
                end
            end
            
            if lowestY then
                position = Vector3.new(position.X, lowestY - 2, position.Z)
                velocity = velocity * Vector3.new(1, -1, 1)
            end
        end
        
        speed = #touchingBottom <= 0 and #touchingTop <= 0 and 142 or speed
        elapsed = elapsed + step
    end
    
    for _, part in pairs(simParts) do
        part:Destroy()
    end
    
    return position + Vector3.new(0, simDiv, 0)
end

-- METODE 2: ELECTRO
-- ============================================================
function SuperPredictor.ElectroPrediction(pos, unit, moveDir, ping)
    local v4474 = unit * 2.05
    local v4475 = v4474.Y
    
    return pos + Vector3.new(
        v4474.X,
        v4475 > 1 and v4475 - 0.15 or (v4475 < -1 and v4475 + 0.15 or (v4475 > 0 and v4475 + 0.05 or (v4475 < 0 and v4475 - 0.05 or 0))),
        v4474.Z
    ) + moveDir * (0.9 + ping / 1500)
end

-- ============================================================
-- METODE 3: RECTUM
-- ============================================================
function SuperPredictor.RectumPrediction(pos, unit, moveDir, ping)
    local pingOffset = ping / 2500
    local v4477 = unit * 2.225
    local v4478 = v4477.X
    local v4479 = v4477.Y - 0.5
    local v4480 = v4477.Z
    
    return pos + Vector3.new(
        v4478 > 0 and v4478 + pingOffset + 0.075 or (v4478 < 0 and (v4478 - pingOffset - 0.075 or 0) or 0),
        v4479 > 0 and v4479 + pingOffset or (v4479 < 0 and (v4479 - pingOffset or 0) or 0),
        v4480 > 0 and v4480 + pingOffset + 0.075 or (v4480 < 0 and v4480 - pingOffset - 0.075 or 0)
    ) + moveDir * 0.8
end

-- METODE 4: REMI (Default)

function SuperPredictor.RemiPrediction(pos, moveDir)
    return pos + moveDir * 0.7
end

-- FUNGSI UTAMA - HYBRID PREDICTION (4-in-1)
-- ============================================================
function SuperPredictor.Predict(target)
    local targetChar = target.Character
    if not targetChar then return nil end
    
    local targetHRP = u613(target)
    if not targetHRP then return nil end
    
    local targetHumanoid = u589(targetChar, 'Humanoid')
    if not targetHumanoid then return nil end
    
    local targetPos = targetHRP.Position
    local targetVel = targetHRP.AssemblyLinearVelocity
    local targetMag = targetVel.Magnitude
    local targetUnit = targetVel.Unit
    local moveDir = targetHumanoid.MoveDirection
    local ping = SuperPredictor.Config.PredictionPing or 100
    
    if targetMag <= 0.19 then
        return targetPos + moveDir * 0.1
    end
    
    local results = {}
    
    -- METODE 1: REVERT
    local revertResult = SuperPredictor.RevertPrediction(
        target, targetHRP, targetHumanoid, 
        targetPos, targetVel, targetUnit, moveDir
    )
    if revertResult then
        table.insert(results, revertResult)
    end
    
    -- METODE 2: ELECTRO
    local electroResult = SuperPredictor.ElectroPrediction(
        targetPos, targetUnit, moveDir, ping
    )
    if electroResult then
        table.insert(results, electroResult)
    end
    
    -- METODE 3: RECTUM
    local rectumResult = SuperPredictor.RectumPrediction(
        targetPos, targetUnit, moveDir, ping
    )
    if rectumResult then
        table.insert(results, rectumResult)
    end
    
    -- METODE 4: REMI
    local remiResult = SuperPredictor.RemiPrediction(
        targetPos, moveDir
    )
    if remiResult then
        table.insert(results, remiResult)
    end
    
    if #results == 0 then
        return targetPos + moveDir * 0.7
    end
    
    local finalPos = Vector3.new(0, 0, 0)
    for _, result in ipairs(results) do
        finalPos = finalPos + result
    end
    finalPos = finalPos / #results
    
    if SuperPredictor.Config.PredictJump and targetHumanoid.Jump then
        finalPos = finalPos + Vector3.new(0, targetHumanoid.JumpPower * 0.3, 0)
    end
    
    local pingOffset = ping / 2000
    finalPos = finalPos + (targetUnit * pingOffset)
    
    return finalPos
end

local function hasLineOfSight(targetChar)
    if not Settings.WallCheck then return true end
    if not lp.Character or not lp.Character:FindFirstChild("HumanoidRootPart") then 
        return false 
    end
    
    local origin = lp.Character.HumanoidRootPart.Position
    local targetPos = targetChar.HumanoidRootPart.Position
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.IgnoreWater = true
    raycastParams.FilterDescendantsInstances = {lp.Character, targetChar, Camera}

    local result = workspace:Raycast(origin, targetPos - origin, raycastParams)
    return result == nil
end

local function getClosestTarget()
    local near = nil
    local minFovDist = math.huge
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, v in pairs(Players:GetPlayers()) do
        if v ~= lp and v.Character and v.Character:FindFirstChild("HumanoidRootPart") then
            local targetHum = v.Character:FindFirstChildOfClass("Humanoid")
            
            if targetHum and targetHum.Health > 0 then
                local hrp = v.Character.HumanoidRootPart
                local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                
                if onScreen then
                    local fovDist = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                    
                    if fovDist <= Settings.FOVRadius then
                        if fovDist < minFovDist then
                            if hasLineOfSight(v.Character) then
                                minFovDist = fovDist
                                near = v
                            end
                        end
                    end
                end
            end
        end
    end
    return near
end

local function getTargetPosition()
    if not currentTarget or not currentTarget.Character then 
        return nil 
    end
    
    local hrp = currentTarget.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    
    local predictedPos = SuperPredictor.Predict(currentTarget)
    
    if not predictedPos then
        predictedPos = hrp.Position
    end
    
    return predictedPos
end

local function setupWeaponHook()
    local success, ws = pcall(function()
        return require(ReplicatedStorage:WaitForChild("ClientServices"):WaitForChild("WeaponService"))
    end)
    
    if success and ws then
        WeaponService = ws
        
        local oldGetTargetPosition = WeaponService.GetTargetPosition
        WeaponService.GetTargetPosition = function(self, ...)
            local pos = getTargetPosition()
            if pos then 
                return CFrame.new(pos) 
            end
            return oldGetTargetPosition(self, ...)
        end

        local oldGetMouseTargetCFrame = WeaponService.GetMouseTargetCFrame
        WeaponService.GetMouseTargetCFrame = function(self, ...)
            local pos = getTargetPosition()
            if pos then 
                return CFrame.new(pos) 
            end
            return oldGetMouseTargetCFrame(self, ...)
        end
        
        if WeaponService.GetThrowTarget then
            local oldGetThrowTarget = WeaponService.GetThrowTarget
            WeaponService.GetThrowTarget = function(self, ...)
                local pos = getTargetPosition()
                if pos then 
                    return CFrame.new(pos) 
                end
                return oldGetThrowTarget(self, ...)
            end
        end
        
        print("✅ Super Predictor v4.0 Hooked!")
        return true
    end
    
    return false
end
