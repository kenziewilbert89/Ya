local Utilities = {}

function Utilities:GetCharacter()
    local char = LocalPlayer.Character
    if char and char.Parent then
        Character = char
        Humanoid = char:FindFirstChildOfClass("Humanoid")
        HumanoidRootPart = char:FindFirstChild("HumanoidRootPart")
        return char
    end
    return nil
end

function Utilities:WaitForCharacter()
    return LocalPlayer.CharacterAdded:Wait()
end

function Utilities:GetClosestPlayer(ignoreTeam)
    local closest = nil
    local closestDist = math.huge
    local localPos = HumanoidRootPart and HumanoidRootPart.Position or Vector3.new()
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            if ignoreTeam and player.Team == LocalPlayer.Team then continue end
            
            local char = player.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                local root = char.HumanoidRootPart
                local dist = (root.Position - localPos).Magnitude
                if dist < closestDist then
                    closestDist = dist
                    closest = player
                end
            end
        end
    end
    return closest
end

function Utilities:GetPlayersByRole(role)
    local result = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local char = player.Character
            if char then
                -- Check role via attributes or game-specific methods
                if char:GetAttribute("Role") == role then
                    table.insert(result, player)
                end
            end
        end
    end
    return result
end

function Utilities:GetRemotes()
    local remoteFolder = ReplicatedStorage:FindFirstChild("Remotes")
    if not remoteFolder then
        remoteFolder = ReplicatedStorage:FindFirstChild("RemoteEvent")
    end
    return remoteFolder
end

function Utilities:FireRemote(remoteName, ...)
    local remotes = Utilities:GetRemotes()
    if remotes then
        local remote = remotes:FindFirstChild(remoteName)
        if remote then
            remote:FireServer(...)
            return true
        end
    end
    return false
end

function Utilities:InvokeRemote(remoteName, ...)
    local remotes = Utilities:GetRemotes()
    if remotes then
        local remote = remotes:FindFirstChild(remoteName)
        if remote and remote:IsA("RemoteFunction") then
            return remote:InvokeServer(...)
        end
    end
    return nil
end

function Utilities:GetDistance(pos1, pos2)
    return (pos1 - pos2).Magnitude
end

function Utilities:IsInRange(pos1, pos2, range)
    return Utilities:GetDistance(pos1, pos2) <= range
end

function Utilities:CreateTween(object, properties, duration, style, direction)
    local tweenInfo = TweenInfo.new(
        duration or 0.5,
        Enum.EasingStyle[style or "Quad"],
        Enum.EasingDirection[direction or "Out"]
    )
    local tween = TweenService:Create(object, tweenInfo, properties)
    return tween
end

function Utilities:Notify(title, message, duration)
    -- Simple notification using a GUI or chat
    game:GetService("StarterGui"):SetCore("SendNotification", {
        Title = title or "Overdrive H",
        Text = message or "",
        Duration = duration or 3
    })
end

function Utilities:GetRandomString(length)
    length = length or 10
    local chars = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
    local result = ""
    for i = 1, length do
        result = result .. chars:sub(math.random(1, #chars), math.random(1, #chars))
    end
    return result
end

local TaskSystem = {
    Tasks = {},
    Running = true
}

function TaskSystem:CreateTask(name, callback, interval)
    if self.Tasks[name] then
        self:RemoveTask(name)
    end
    
    local task = {
        Name = name,
        Callback = callback,
        Running = true,
        Interval = interval or 0.016,
        LastRun = 0,
        Connection = nil
    }
    
    self.Tasks[name] = task
    
    task.Connection = RunService.Heartbeat:Connect(function(deltaTime)
        if not self.Running then return end
        if not task.Running then return end
        
        task.LastRun = task.LastRun + deltaTime
        if task.LastRun >= task.Interval then
            task.LastRun = 0
            local success, err = pcall(task.Callback)
            if not success and Config.DebugMode then
                warn("Task error [" .. name .. "]: " .. tostring(err))
            end
        end
    end)
    
    return task
end

function TaskSystem:RemoveTask(name)
    local task = self.Tasks[name]
    if task then
        task.Running = false
        if task.Connection then
            task.Connection:Disconnect()
        end
        self.Tasks[name] = nil
        return true
    end
    return false
end

function TaskSystem:ToggleTask(name, enabled, callback, interval)
    if enabled then
        return self:CreateTask(name, callback, interval)
    else
        return self:RemoveTask(name)
    end
end

function TaskSystem:IsTaskRunning(name)
    return self.Tasks[name] ~= nil and self.Tasks[name].Running
end

function TaskSystem:GetAllTasks()
    local result = {}
    for name, _ in pairs(self.Tasks) do
        table.insert(result, name)
    end
    return result
end

-- Global shortcuts
function MakeTask(name, callback, interval) return TaskSystem:CreateTask(name, callback, interval) end
function RemoveTask(name) return TaskSystem:RemoveTask(name) end
function IsTaskRunning(name) return TaskSystem:IsTaskRunning(name) end

local Combat = {}

-- 7.1 Auto Parry
function Combat:AutoParry()
    MakeTask("AutoParry", function()
        if not GetProperty("AutoParry") then return end
        
        local char = Utilities:GetCharacter()
        if not char or not Humanoid then return end
        
        -- Check if clashing
        local isClashing = char:GetAttribute("Clashing") or false
        
        -- If using legit mode
        if GetProperty("AutoParryLegitMode") then
            -- Random delay before parrying
            local delay = math.random(100, 300) / 1000
            task.wait(delay)
        end
        
        -- Check for incoming attacks
        -- Implementation depends on game
        if CurrentGame == "Murder Mystery 2" then
            self:MM2Parry()
        elseif CurrentGame == "Blade Ball" then
            self:BladeBallParry()
        end
    end)
end

function Combat:MM2Parry()
    -- MM2 specific parry logic
    local remotes = Utilities:GetRemotes()
    if remotes then
        local parryRemote = remotes:FindFirstChild("Parry")
        if parryRemote then
            parryRemote:FireServer()
        end
    end
end

function Combat:BladeBallParry()
    -- Blade Ball specific parry logic
    -- Check for ball approaching
    local ball = workspace:FindFirstChild("Ball")
    if ball then
        local char = Utilities:GetCharacter()
        if char and HumanoidRootPart then
            local distance = (ball.Position - HumanoidRootPart.Position).Magnitude
            if distance < 30 then
                -- Parry the ball
                local remotes = Utilities:GetRemotes()
                if remotes then
                    local parryRemote = remotes:FindFirstChild("Parry")
                    if parryRemote then
                        parryRemote:FireServer()
                    end
                end
            end
        end
    end
end

-- 7.2 Auto Forcefield
function Combat:AutoForcefield()
    MakeTask("AutoForcefield", function()
        if not GetProperty("AutoForcefield") then return end
        
        local char = Utilities:GetCharacter()
        if not char then return end
        
        -- Check if clashing
        local isClashing = char:GetAttribute("Clashing") or false
        
        if isClashing and not char:FindFirstChild("ForceField") then
            -- Activate forcefield
            local forcefield = Instance.new("ForceField")
            forcefield.Parent = char
            
            -- Remove after clash ends
            task.wait(0.5)
            if forcefield and forcefield.Parent then
                forcefield:Destroy()
            end
        end
    end)
end

local SilentAim = {}

function SilentAim:GetPredictionType()
    return GetProperty("GunSilentAimPrediction") or "Revert"
end

function SilentAim:GetPredictionAccuracy(predictionType)
    local accuracies = {
        Revert = 0.966,
        Rectum = 0.962,
        Electro = 0.96,
        Remi = 0.87
    }
    return accuracies[predictionType] or 0.9
end

function SilentAim:CalculatePrediction(target, predictionType)
    local char = Utilities:GetCharacter()
    if not char or not HumanoidRootPart then return nil end
    
    local targetRoot = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
    if not targetRoot then return nil end
    
    local targetPos = targetRoot.Position
    local myPos = HumanoidRootPart.Position
    
    -- Base prediction
    local prediction = targetPos
    
    -- Get prediction settings
    local ping = GetProperty("PredictionPing") or 100
    local verticalMult = GetProperty("PredictionVerticalMultiplier") or 100
    local horizontalMult = GetProperty("PredictionHorizontalMultiplier") or 100
    local interval = GetProperty("PredictionInterval") or 20
    local wallCheck = GetProperty("PredictionWallCheck")
    
    -- Apply prediction based on type
    if predictionType == "Revert" then
        -- Basic prediction
        local velocity = targetRoot.Velocity or Vector3.new()
        local time = ping / 1000
        prediction = targetPos + velocity * time
        
    elseif predictionType == "Rectum" then
        -- Advanced prediction with vertical/horizontal multipliers
        local velocity = targetRoot.Velocity or Vector3.new()
        local time = ping / 1000
        
        local horizontal = Vector3.new(velocity.X, 0, velocity.Z) * (horizontalMult / 100)
        local vertical = Vector3.new(0, velocity.Y, 0) * (verticalMult / 100)
        
        prediction = targetPos + (horizontal + vertical) * time
        
    elseif predictionType == "Electro" then
        -- Electro prediction with interval
        local velocity = targetRoot.Velocity or Vector3.new()
        local time = ping / 1000 * (interval / 20)
        
        prediction = targetPos + velocity * time
        
    elseif predictionType == "Remi" then
        -- Remi prediction with ping compensation
        local velocity = targetRoot.Velocity or Vector3.new()
        local time = ping / 1000 * 1.2
        
        prediction = targetPos + velocity * time
    end
    
    -- Wall check
    if wallCheck then
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Blacklist
        params.FilterDescendantsInstances = {char}
        
        local ray = workspace:Raycast(myPos, (prediction - myPos).Unit * 200, params)
        if ray then
            -- If wall in way, adjust prediction
            prediction = ray.Position
        end
    end
    
    -- Predict jump
    if GetProperty("PredictJump") and targetRoot.Velocity.Y > 0 then
        local velocity = targetRoot.Velocity or Vector3.new()
        local time = ping / 1000
        local gravity = workspace.Gravity
        prediction = prediction + Vector3.new(0, velocity.Y * time + 0.5 * gravity * time^2, 0)
    end
    
    return prediction
end

function SilentAim:GetAimAngle(target)
    local char = Utilities:GetCharacter()
    if not char or not HumanoidRootPart then return nil end
    
    local targetRoot = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
    if not targetRoot then return nil end
    
    local targetPos = targetRoot.Position
    local myPos = HumanoidRootPart.Position
    
    -- Get prediction
    local predictionType = self:GetPredictionType()
    local predictedPos = self:CalculatePrediction(target, predictionType)
    
    if not predictedPos then return nil end
    
    -- Calculate angle
    local direction = (predictedPos - myPos).Unit
    local angle = math.atan2(direction.Y, direction.X)
    
    return angle
end

    MakeTask("HitboxExpander", function()

        local size = GetProperty("HitboxSize") or 2
        local transparency = GetProperty("HitboxTransparency") or 70
        
        local r = GetProperty("HitboxColorR") or 250
        local g = GetProperty("HitboxColorG") or 250
        local b = GetProperty("HitboxColorB") or 250
        
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                local char = player.Character
                if char then
                    -- Find hitbox parts
                    for _, part in ipairs(char:GetDescendants()) do
                        if part:IsA("BasePart") and part.Name == "HumanoidRootPart" then
                            -- Create expanded hitbox
                            local hitbox = part:FindFirstChild("ExpandedHitbox")
                            if not hitbox then
                                hitbox = Instance.new("Part")
                                hitbox.Name = "ExpandedHitbox"
                                hitbox.Size = Vector3.new(size, size, size)
                                hitbox.Transparency = transparency / 100
                                hitbox.BrickColor = BrickColor.new(Color3.fromRGB(r, g, b))
                                hitbox.Anchored = false
                                hitbox.CanCollide = false
                                hitbox.Parent = part
                                
                                -- Weld to root part
                                local weld = Instance.new("Weld")
                                weld.Part0 = part
                                weld.Part1 = hitbox
                                weld.C0 = CFrame.new(0, 0, 0)
                                weld.Parent = hitbox
                            end
                        end
                    end
                end
            end
        end
    end, 0.5)
end

        local flySpeed = GetProperty("FlySpeed") or 15
        
        -- Create body velocity if not exists
        if not bodyVelocity then
            bodyVelocity = Instance.new("BodyVelocity")
            bodyVelocity.MaxForce = Vector3.new(4000, 4000, 4000)
            bodyVelocity.Parent = HumanoidRootPart
        end
        
        if not bodyGyro then
            bodyGyro = Instance.new("BodyGyro")
            bodyGyro.MaxTorque = Vector3.new(4000, 4000, 4000)
            bodyGyro.Parent = HumanoidRootPart
        end
        
        -- Get movement input
        local moveDirection = Vector3.new()
        
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then
            moveDirection = moveDirection + HumanoidRootPart.CFrame.LookVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then
            moveDirection = moveDirection - HumanoidRootPart.CFrame.LookVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then
            moveDirection = moveDirection - HumanoidRootPart.CFrame.RightVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then
            moveDirection = moveDirection + HumanoidRootPart.CFrame.RightVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
            moveDirection = moveDirection + Vector3.new(0, 1, 0)
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
            moveDirection = moveDirection - Vector3.new(0, 1, 0)
        end
        
        if moveDirection.Magnitude > 0 then
            bodyVelocity.Velocity = moveDirection.Unit * flySpeed
        else
            bodyVelocity.Velocity = Vector3.new(0, 0, 0)
        end
        
        bodyGyro.CFrame = HumanoidRootPart.CFrame
        
        -- Disable gravity
        if Humanoid then
            Humanoid.PlatformStand = true
        end
    end, 0.016)
end

function AutoFarm:CollectCoins()
    local char = Utilities:GetCharacter()
    if not char or not HumanoidRootPart then return end
    
    -- Find nearest coin
    local nearestCoin = nil
    local nearestDist = math.huge
    
    for _, part in ipairs(workspace:GetDescendants()) do
        if part:IsA("BasePart") and part.Name == "Coin" then
            local dist = (part.Position - HumanoidRootPart.Position).Magnitude
            if dist < nearestDist then
                nearestDist = dist
                nearestCoin = part
            end
        end
    end
    
    if nearestCoin and nearestDist < 50 then
        -- Move to coin
        char:MoveTo(nearestCoin.Position)
        
        -- Collect coin
        if nearestDist < 5 then
            local remotes = Utilities:GetRemotes()
            if remotes then
                local collectRemote = remotes:FindFirstChild("CollectCoin")
                if collectRemote then
                    collectRemote:FireServer(nearestCoin)
                end
            end
        end
    end
end

function GUI:SwitchTab(tabName, tabButtons, contentContainer)
    -- Clear content
    for _, child in ipairs(contentContainer:GetChildren()) do
        child:Destroy()
    end
    
    -- Update tab buttons
    for name, btn in pairs(tabButtons) do
        if name == tabName then
            btn.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        else
            btn.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
            btn.TextColor3 = Color3.fromRGB(200, 200, 200)
        end
    end
    
    -- Create content based on tab
    local yPos = 5
    
    if tabName == "Main" then
        yPos = GUI:AddMainTab(contentContainer, yPos)
    elseif tabName == "Combat" then
        yPos = GUI:AddCombatTab(contentContainer, yPos)
    elseif tabName == "Visuals" then
        yPos = GUI:AddVisualTab(contentContainer, yPos)
    elseif tabName == "SelfMods" then
        yPos = GUI:AddSelfModsTab(contentContainer, yPos)
    elseif tabName == "AutoFarm" then
        yPos = GUI:AddAutoFarmTab(contentContainer, yPos)
    elseif tabName == "MM2" then
        yPos = GUI:AddMM2Tab(contentContainer, yPos)
    elseif tabName == "Settings" then
        yPos = GUI:AddSettingsTab(contentContainer, yPos)
    end
    
    -- Update canvas size
    contentContainer.CanvasSize = UDim2.new(0, 0, 0, yPos + 10)
end

    local percent = (currentValue - min) / (max - min)
    fill.Size = UDim2.new(percent, 0, 1, 0)
    
    slider.MouseButton1Down:Connect(function()
        local connection
        connection = UserInputService.InputChanged:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseMovement then
                local mousePos = UserInputService:GetMouseLocation()
                local sliderPos = slider.AbsolutePosition
                local sliderSize = slider.AbsoluteSize
                
                local x = (mousePos.X - sliderPos.X) / sliderSize.X
                x = math.clamp(x, 0, 1)
                
                local newValue = math.round(min + (max - min) * x)
                SetProperty(property, newValue)
                
                valueText.Text = tostring(newValue)
                fill.Size = UDim2.new(x, 0, 1, 0)
            end
        end)
        
        UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 then
                connection:Disconnect()
            end
        end)
    end)

function GUI:MakeDraggable(frame)
    local dragging = false
    local dragInput = nil
    local dragStart = nil
    local startPos = nil
    
    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    
    frame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement and dragging then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end
