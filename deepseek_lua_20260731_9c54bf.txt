-- ============================================================
-- UTILITY FUNCTIONS
-- ============================================================

local function GetRoot(plr)
    if not plr or not plr.Character then return nil end
    return plr.Character:FindFirstChild("HumanoidRootPart")
end

local function GetHumanoid(plr)
    if not plr or not plr.Character then return nil end
    return plr.Character:FindFirstChildOfClass("Humanoid")
end

local function IsAlive(plr)
    local hum = GetHumanoid(plr)
    return hum and hum.Health > 0
end

local function Distance(pos1, pos2)
    return (pos1 - pos2).Magnitude
end

local function GetAllPlayers()
    return Players:GetPlayers()
end

local function GetPlayerByName(name)
    for _, plr in ipairs(GetAllPlayers()) do
        if plr.Name:lower():match(name:lower()) then
            return plr
        end
    end
    return nil
end

-- ============================================================
-- GET TARGET BY CLOSEST
-- ============================================================

function GetTargetByClosest()
    local char = LocalPlayer.Character
    if not char then return nil end
    
    local myRoot = char:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    
    local closest = nil
    local closestDist = math.huge
    
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and IsAlive(plr) then
            local root = GetRoot(plr)
            if root then
                local dist = Distance(myRoot.Position, root.Position)
                if dist < closestDist then
                    closestDist = dist
                    closest = plr
                end
            end
        end
    end
    
    return closest
end

-- ============================================================
-- GET TARGET BY MOUSE
-- ============================================================

function GetTargetByMouse()
    local mouse = LocalPlayer:GetMouse()
    if not mouse then return nil end
    
    local closest = nil
    local closestDist = math.huge
    
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and IsAlive(plr) then
            local root = GetRoot(plr)
            if root then
                local screenPos, onScreen = Camera:WorldToViewportPoint(root.Position)
                if onScreen then
                    local mousePos = Vector2.new(mouse.X, mouse.Y)
                    local screenVec = Vector2.new(screenPos.X, screenPos.Y)
                    local dist = (mousePos - screenVec).Magnitude
                    if dist < closestDist then
                        closestDist = dist
                        closest = plr
                    end
                end
            end
        end
    end
    
    return closest
end

-- ============================================================
-- PREDIKSI POSISI KNIFE
-- ============================================================

function GetKnifePrediction(target)
    if not target then return nil end
    
    local targetRoot = GetRoot(target)
    if not targetRoot then return nil end
    
    local velocity = targetRoot.AssemblyLinearVelocity
    local predTime = getgenv().KnifePredictionTime or 0.16
    local predictedPos = targetRoot.Position + Vector3.new(velocity.X, 0, velocity.Z) * predTime
    
    return predictedPos
end

-- ============================================================
-- KILL PLAYER (DENGAN HandleTouched - THUNDER HUB REMOTE)
-- ============================================================

function KillPlayer(targetPlr)
    if not targetPlr or not IsAlive(targetPlr) then return false end
    
    local char = LocalPlayer.Character
    if not char then return false end
    
    -- Equip knife
    local knife = EquipKnife()
    if not knife then
        knife = GetKnife()
    end
    if not knife then return false end
    
    local targetRoot = GetRoot(targetPlr)
    if not targetRoot then return false end
    
    -- Anchor target
    local myRoot = char:FindFirstChild("HumanoidRootPart")
    if myRoot then
        targetRoot.Anchored = true
        targetRoot.CFrame = myRoot.CFrame * CFrame.new(0, 0, -2)
    end
    
    task.wait(0.05)
    
    local killed = false
    
    -- METHOD 1: HandleTouched (THUNDER HUB)
    pcall(function()
        local knifeEvents = char:FindFirstChild("Knife")
        if knifeEvents then
            knifeEvents = knifeEvents:FindFirstChild("Events")
            if knifeEvents then
                local handleTouched = knifeEvents:FindFirstChild("HandleTouched")
                if handleTouched then
                    handleTouched:FireServer(targetRoot)
                    killed = true
                end
            end
        end
    end)
    
    -- METHOD 2: KnifeStabbed (BACKUP)
    if not killed then
        pcall(function()
            local knifeEvents = char:FindFirstChild("Knife")
            if knifeEvents then
                knifeEvents = knifeEvents:FindFirstChild("Events")
                if knifeEvents then
                    local knifeStabbed = knifeEvents:FindFirstChild("KnifeStabbed")
                    if knifeStabbed then
                        knifeStabbed:FireServer()
                        killed = true
                    end
                end
            end
        end)
    end
    
    -- METHOD 3: firetouchinterest (BACKUP)
    if not killed and knife and knife:FindFirstChild("Handle") then
        pcall(function()
            firetouchinterest(targetRoot, knife.Handle, 1)
            firetouchinterest(targetRoot, knife.Handle, 0)
            killed = true
        end)
    end
    
    -- Unanchor target
    task.wait(0.05)
    targetRoot.Anchored = false
    
    return killed
end

-- ============================================================
-- KILL ALL PLAYERS
-- ============================================================

function KillAllPlayers()
    local killed = 0
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and IsAlive(plr) then
            if KillPlayer(plr) then
                killed = killed + 1
            end
            task.wait(0.05)
        end
    end
    return killed
end

-- ============================================================
-- KILL SHERIFF
-- ============================================================

function KillSheriff()
    local sheriff = GetSheriff()
    if sheriff and IsAlive(sheriff) then
        return KillPlayer(sheriff)
    end
    return false
end

-- ============================================================
-- SHOOT MURDERER (DENGAN REMOTE YANG BENAR)
-- ============================================================

function ShootMurderer()
    -- CEK CHARACTER
    local char = LocalPlayer.Character
    if not char then
        WindUI:Notify({Title = "❌ Error", Content = "Character not found!", Duration = 2})
        return false
    end
    
    -- CEK GUN
    local gun = EquipGun()
    if not gun then
        gun = GetGun()
    end
    
    if not gun then
        WindUI:Notify({Title = "❌ No Gun", Content = "Are you Sheriff?", Duration = 3})
        return false
    end
    
    -- CEK MURDERER
    local murderer = GetMurderer()
    if not murderer then
        WindUI:Notify({Title = "❌ No Target", Content = "No Murderer found!", Duration = 2})
        return false
    end
    
    if not murderer.Character then
        WindUI:Notify({Title = "❌ Error", Content = "Murderer has no character!", Duration = 2})
        return false
    end
    
    local targetRoot = GetRoot(murderer)
    if not targetRoot then
        WindUI:Notify({Title = "❌ Error", Content = "Target root not found!", Duration = 2})
        return false
    end
    
    local myRoot = char:FindFirstChild("HumanoidRootPart")
    if not myRoot then
        WindUI:Notify({Title = "❌ Error", Content = "My root not found!", Duration = 2})
        return false
    end
    
    -- PREDIKSI POSISI
    local mode = getgenv().SheriffSilentAimMode or "Dynamic"
    local targetPos = nil
    
    if mode == "Overflow (Premium)" then
        local rootPos = targetRoot.Position
        local moveDir = murderer.Character.Humanoid and murderer.Character.Humanoid.MoveDirection or Vector3.new()
        local velocity = targetRoot.AssemblyLinearVelocity
        targetPos = rootPos + moveDir + (velocity.Magnitude > 0 and velocity.Unit * 2.12565 or Vector3.new(0, 0, 0))
        
    elseif mode == "Dynamic" then
        local rootCF = targetRoot.CFrame
        local moveDir = murderer.Character.Humanoid and murderer.Character.Humanoid.MoveDirection or Vector3.new()
        local velocity = targetRoot.AssemblyLinearVelocity
        local speed = velocity.Magnitude / 15
        targetPos = (rootCF + moveDir * speed).Position
        
    elseif mode == "Velocity" then
        local velocity = targetRoot.AssemblyLinearVelocity
        local predTime = getgenv().GunPredictionTime or 0.125
        targetPos = targetRoot.Position + (velocity * predTime)
        
    else  -- Regular
        local rootCF = targetRoot.CFrame
        local moveDir = murderer.Character.Humanoid and murderer.Character.Humanoid.MoveDirection or Vector3.new()
        targetPos = (rootCF + moveDir).Position
    end
    
    -- Ambil attachment untuk CFrame
    local attachment = myRoot:FindFirstChild("GunRaycastAttachment")
    local shootCFrame = attachment and attachment.WorldCFrame or myRoot.CFrame
    
    -- SHOOT (PAKAI REMOTE YANG BENAR)
    local success = false
    
    pcall(function()
        -- Method 1: Gun.Shoot
        if gun:FindFirstChild("Shoot") then
            gun.Shoot:FireServer(shootCFrame, CFrame.new(targetPos))
            success = true
        end
    end)
    
    if not success then
        pcall(function()
            -- Method 2: Cari remote di ReplicatedStorage
            local remotes = ReplicatedStorage:FindFirstChild("Remotes")
            if remotes then
                local shootRemote = remotes:FindFirstChild("Extras"):FindFirstChild("ShootGun")
                if shootRemote then
                    shootRemote:FireServer(shootCFrame, CFrame.new(targetPos))
                    success = true
                end
            end
        end)
    end
    
    if not success then
        pcall(function()
            -- Method 3: Cari remote langsung
            local remote = ReplicatedStorage:FindFirstChild("Shoot", true)
            if remote then
                remote:FireServer(shootCFrame, CFrame.new(targetPos))
                success = true
            end
        end)
    end
    
    if success then
        WindUI:Notify({Title = "✅ Shot fired!", Content = "Target: " .. murderer.Name, Duration = 2})
        return true
    else
        WindUI:Notify({Title = "❌ Failed", Content = "Failed to shoot!", Duration = 3})
        return false
    end
end

-- ============================================================
-- THROW KNIFE (DENGAN REMOTE YANG BENAR)
-- ============================================================

function ThrowKnife()
    -- CEK MURDERER
    if not IsMurderer() then
        WindUI:Notify({Title = "❌ Not Murderer", Content = "You don't have a knife!", Duration = 2})
        return false
    end
    
    -- CEK CHARACTER
    local char = LocalPlayer.Character
    if not char then
        WindUI:Notify({Title = "❌ Error", Content = "Character not found!", Duration = 2})
        return false
    end
    
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid then
        WindUI:Notify({Title = "❌ Error", Content = "Humanoid not found!", Duration = 2})
        return false
    end
    
    -- CEK KNIFE
    local knife = EquipKnife()
    if not knife then
        knife = GetKnife()
    end
    
    if not knife then
        WindUI:Notify({Title = "❌ No Knife", Content = "You need a knife to throw!", Duration = 2})
        return false
    end
    
    -- GET TARGET
    local target = GetKnifeTarget()
    if not target then
        WindUI:Notify({Title = "❌ No Target", Content = "No alive players found!", Duration = 3})
        return false
    end
    
    if not target.Character then
        WindUI:Notify({Title = "❌ Error", Content = "Target has no character!", Duration = 2})
        return false
    end
    
    if not IsAlive(target) then
        WindUI:Notify({Title = "❌ Target Dead", Content = target.Name .. " is already dead!", Duration = 2})
        return false
    end
    
    -- PREDIKSI POSISI
    local targetRoot = GetRoot(target)
    if not targetRoot then
        WindUI:Notify({Title = "❌ Error", Content = "Target root not found!", Duration = 2})
        return false
    end
    
    local predictedPos = GetKnifePrediction(target)
    if not predictedPos then
        predictedPos = targetRoot.Position
    end
    
    -- THROW KNIFE
    local success = false
    
    -- METHOD 1: KnifeThrown (Events)
    pcall(function()
        local knifeEvents = char:FindFirstChild("Knife")
        if knifeEvents then
            knifeEvents = knifeEvents:FindFirstChild("Events")
            if knifeEvents then
                local knifeThrown = knifeEvents:FindFirstChild("KnifeThrown")
                if knifeThrown then
                    local handleCFrame = knife.Handle and knife.Handle.CFrame or CFrame.new()
                    knifeThrown:FireServer(handleCFrame, CFrame.new(predictedPos))
                    success = true
                end
            end
        end
    end)
    
    -- METHOD 2: ReplicatedStorage
    if not success then
        pcall(function()
            local remote = ReplicatedStorage:FindFirstChild("KnifeThrown", true)
            if remote then
                local handleCFrame = knife.Handle and knife.Handle.CFrame or CFrame.new()
                remote:FireServer(handleCFrame, CFrame.new(predictedPos))
                success = true
            end
        end)
    end
    
    -- METHOD 3: Throw (Tool)
    if not success then
        pcall(function()
            local knifeEvents = char:FindFirstChild("Knife")
            if knifeEvents then
                knifeEvents = knifeEvents:FindFirstChild("Events")
                if knifeEvents then
                    local knifeThrow = knifeEvents:FindFirstChild("Throw")
                    if knifeThrow then                        knifeThrow:FireServer(CFrame.new(predictedPos))
                        success = true
                    end
                end
            end
        end)
    end
    
    -- METHOD 4: KnifeStabbed (BACKUP)
    if not success then
        pcall(function()
            local knifeEvents = char:FindFirstChild("Knife")
            if knifeEvents then
                knifeEvents = knifeEvents:FindFirstChild("Events")
                if knifeEvents then
                    local knifeStabbed = knifeEvents:FindFirstChild("KnifeStabbed")
                    if knifeStabbed then
                        knifeStabbed:FireServer()
                        success = true
                    end
                end
            end
        end)
    end
    
    if success then
        WindUI:Notify({Title = "✅ Knife Thrown!", Content = "Target: " .. target.Name, Duration = 2})
        return true
    else
        WindUI:Notify({Title = "❌ Throw Failed", Content = "Failed to throw knife!", Duration = 3})
        return false
    end
end

local autoShootRunning = false
local autoShootConnection = nil
local autoShootRadius = 14
local autoShootAngle = 0

function ToggleAutoShoot(state)
    if state then
        if autoShootRunning then return end
        autoShootRunning = true
        
        autoShootConnection = RunService.Heartbeat:Connect(function()
            if not autoShootRunning then return end
            
            -- CEK SHERIFF
            if not IsSheriff() then return end
            
            local murderer = GetMurderer()
            if not murderer or not murderer.Character then return end
            
            local targetRoot = GetRoot(murderer)
            if not targetRoot then return end
            
            local char = LocalPlayer.Character
            if not char then return end
            
            local gun = GetGun()
            if not gun then
                gun = EquipGun()
            end
            if not gun then return end
            
            local myRoot = char:FindFirstChild("HumanoidRootPart")
            if not myRoot then return end
            
            -- PREDIKSI
            local mode = getgenv().SheriffSilentAimMode or "Dynamic"
            local targetPos = nil
            
            if mode == "Overflow (Premium)" then
                local rootPos = targetRoot.Position
                local moveDir = murderer.Character.Humanoid and murderer.Character.Humanoid.MoveDirection or Vector3.new()
                local velocity = targetRoot.AssemblyLinearVelocity
                targetPos = rootPos + moveDir + (velocity.Magnitude > 0 and velocity.Unit * 2.12565 or Vector3.new(0, 0, 0))
                
            elseif mode == "Dynamic" then
                local rootCF = targetRoot.CFrame
                local moveDir = murderer.Character.Humanoid and murderer.Character.Humanoid.MoveDirection or Vector3.new()
                local velocity = targetRoot.AssemblyLinearVelocity
                local speed = velocity.Magnitude / 15
                targetPos = (rootCF + moveDir * speed).Position
                
            elseif mode == "Velocity" then
                local velocity = targetRoot.AssemblyLinearVelocity
                local predTime = getgenv().GunPredictionTime or 0.125
                targetPos = targetRoot.Position + (velocity * predTime)
                
            else
                local rootCF = targetRoot.CFrame
                local moveDir = murderer.Character.Humanoid and murderer.Character.Humanoid.MoveDirection or Vector3.new()
                targetPos = (rootCF + moveDir).Position
            end
            
            if targetPos then
                local attachment = myRoot:FindFirstChild("GunRaycastAttachment")
                local shootCFrame = attachment and attachment.WorldCFrame or myRoot.CFrame
                
                pcall(function()
                    if gun:FindFirstChild("Shoot") then
                        gun.Shoot:FireServer(shootCFrame, CFrame.new(targetPos))
                    end
                end)
            end
        end)
        
        -- CIRCLE MOVEMENT
        task.spawn(function()
            while autoShootRunning do
                if not IsSheriff() then
                    task.wait(1)
                else
                    local char = LocalPlayer.Character
                    if char then
                        local murderer = GetMurderer()
                        if murderer then
                            local targetRoot = GetRoot(murderer)
                            if targetRoot then
                                local x = math.cos(math.rad(autoShootAngle)) * autoShootRadius
                                local z = math.sin(math.rad(autoShootAngle)) * autoShootRadius
                                char:SetPrimaryPartCFrame(targetRoot.CFrame * CFrame.new(x, 0, z))
                                autoShootAngle = (autoShootAngle + 5) % 360
                                task.wait(0.03)
                            end
                        else
                            task.wait(0.4)
                        end
                    end
                end
            end
        end)
        
    else
        if autoShootConnection then
            autoShootConnection:Disconnect()
            autoShootConnection = nil
        end
        autoShootRunning = false
    end
end

function SetAutoShootRadius(value)
    autoShootRadius = tonumber(value) or 14
end

        if knifeAuraRunning then return end

        knifeAuraConnection = RunService.Heartbeat:Connect(function()
            if not knifeAuraRunning then return end
            
            if IsMurderer() then

                if myRoot then
                    for _, plr in ipairs(Players:GetPlayers()) do
                        if plr ~= LocalPlayer and IsAlive(plr) then

                            if root and Distance(myRoot.Position, root.Position) <= knifeAuraRange then

            end

        if knifeAuraConnection then
            knifeAuraConnection:Disconnect()

local speedGlitchRunning = false
local speedGlitchIntensity = 35

function ToggleSpeedGlitch(state)
    if state then
        if speedGlitchRunning then return end
        speedGlitchRunning = true
        
        UserInputService.JumpRequest:Connect(function()
            if not speedGlitchRunning then return end
            
            local char = LocalPlayer.Character
            if not char then return end
            
            local humanoid = char:FindFirstChildOfClass("Humanoid")
            if not humanoid then return end
            
            local origSpeed = humanoid.WalkSpeed
            humanoid.WalkSpeed = speedGlitchIntensity
            
            local state = humanoid:GetState()
            local waitTime = 0
            
            while state == Enum.HumanoidStateType.Landed or 
                  state == Enum.HumanoidStateType.Freefall or 
                  state == Enum.HumanoidStateType.Jumping do
                task.wait()
                state = humanoid:GetState()
                waitTime = waitTime + 1
                
                if waitTime > 50 then
                    break
                end
            end
            
            humanoid.WalkSpeed = origSpeed
        end)
        
    else
        speedGlitchRunning = false
    end
end

function SetSpeedGlitchIntensity(value)
    speedGlitchIntensity = tonumber(value) or 35
end

local fakeKnifeTool = nil

function CreateFakeKnife()
    if fakeKnifeTool then
        WindUI:Notify({Title = "Fake Knife", Content = "Already equipped!", Duration = 2})
        return
    end
    
    local char = LocalPlayer.Character
    if not char then
        WindUI:Notify({Title = "Error", Content = "Character not found!", Duration = 2})
        return
    end
    
    -- Cari KnifeDisplay
    local knifeDisplay = nil
    for _, child in ipairs(char:GetChildren()) do
        if child.Name == "KnifeDisplay" then
            knifeDisplay = child
            break
        end
    end
    
    if not knifeDisplay then
        local playerChar = Workspace:FindFirstChild(LocalPlayer.Name)
        if playerChar then
            for _, child in ipairs(playerChar:GetChildren()) do
                if child.Name == "KnifeDisplay" then
                    knifeDisplay = child
                    break
                end
            end
        end
    end
    
    if not knifeDisplay then
        WindUI:Notify({Title = "Error", Content = "KnifeDisplay not found!", Duration = 2})
        return
    end
    
    -- Buat Tool
    local tool = Instance.new("Tool")
    tool.Name = "Fake Knife"
    tool.RequiresHandle = true
    tool.CanBeDropped = false
    tool.GripPos = Vector3.new(0, -1.15, 0)
    
    -- Handle
    local handle = Instance.new("Part")
    handle.Name = "Handle"
    handle.Size = knifeDisplay.Size or Vector3.new(0.3, 3.4, 1.1)
    handle.CanCollide = false
    handle.Anchored = false
    handle.Parent = tool
    
    -- Clone mesh
    local mesh = knifeDisplay:FindFirstChild("Mesh")
    if mesh then
        local newMesh = mesh:Clone()
        newMesh.Parent = handle
    end
    
    -- Animasi
    local anim1 = Instance.new("Animation")
    anim1.AnimationId = "rbxassetid://2467567750"
    local anim2 = Instance.new("Animation")
    anim2.AnimationId = "rbxassetid://1957890538"
    local anims = {anim1, anim2}
    
    local isStabbing = false
    tool.Activated:Connect(function()
        if not isStabbing then
            isStabbing = true
            local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                local animTrack = humanoid:LoadAnimation(anims[math.random(1, 2)])
                animTrack:Play()
                task.wait(1)
                isStabbing = false
            else
                isStabbing = false
            end
        end
    end)
    
    tool.Parent = LocalPlayer.Backpack
    fakeKnifeTool = tool
    WindUI:Notify({Title = "Fake Knife", Content = "Equipped!", Duration = 2})
end

function DestroyFakeKnife()
    if fakeKnifeTool then
        fakeKnifeTool:Destroy()
        fakeKnifeTool = nil
        WindUI:Notify({Title = "Fake Knife", Content = "Removed!", Duration = 2})
    end
end

local grabLock = false

function GrabGun()
    if grabLock then 
        WindUI:Notify({Title = "Grab Gun", Content = "Please wait...", Duration = 2})
        return false
    end
    
    grabLock = true
    
    -- CEK MURDERER
    if IsMurderer() then
        WindUI:Notify({Title = "Grab Gun", Content = "You are the Murderer!", Duration = 2})
        grabLock = false
        return false
    end
    
    -- CARI GUNDROP
    local GunDrop = nil
    for _, Descendant in ipairs(Workspace:GetDescendants()) do
        if Descendant.Name == "GunDrop" and Descendant:IsA("BasePart") then
            GunDrop = Descendant
            break
        end
    end
    
    if not GunDrop then
        WindUI:Notify({Title = "Grab Gun", Content = "No gun drop found!", Duration = 2})
        grabLock = false
        return false
    end
    
    local char = LocalPlayer.Character
    if not char then
        grabLock = false
        return false
    end
    
    local RootPart = char:FindFirstChild("HumanoidRootPart")
    if not RootPart then
        grabLock = false
        return false
    end
    
    -- ANTI FLING
    local OriginalPos = RootPart.Position
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    
    if humanoid then
        humanoid.PlatformStand = true
    end
    
    RootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    RootPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
    RootPart.CFrame = GunDrop.CFrame + Vector3.new(0, 2, 0)
    
    task.wait(0.15)
    
    RootPart.CFrame = CFrame.new(OriginalPos.X, OriginalPos.Y + 5, OriginalPos.Z)
    RootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    
    task.wait(0.05)
    
    if humanoid then
        humanoid.PlatformStand = false
    end
    
    WindUI:Notify({Title = "Grab Gun", Content = "Gun acquired!", Duration = 2})
    grabLock = false
    return true
end

function BreakGun()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local gun = plr.Character:FindFirstChild("Gun")
            if gun then
                local knifeServer = gun:FindFirstChild("KnifeServer")
                if knifeServer then
                    local shootGun = knifeServer:FindFirstChild("ShootGun")
                    if shootGun then
                        pcall(function()
                            shootGun:InvokeServer(0, {}, "AH2")
                            WindUI:Notify({Title = "Break Gun", Content = "Gun broken!", Duration = 2})
                            return true
                        end)
                    end
                end
            end
        end
    end
    
    WindUI:Notify({Title = "Break Gun", Content = "No gun to break!", Duration = 2})
    return false
end

-- STEAL GUN
-- ============================================================

function StealGun()
    -- Cari sheriff
    local sheriff = nil
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local gun = plr.Character:FindFirstChild("Gun")
            if gun then
                sheriff = plr
                break
            end
        end
    end
    
    if not sheriff then
        WindUI:Notify({Title = "Steal Gun", Content = "No Sheriff found!", Duration = 2})
        return false
    end
    
    -- Fling sheriff (anti fling method)
    local targetRoot = GetRoot(sheriff)
    local myRoot = GetRoot(LocalPlayer)
    
    if targetRoot and myRoot then
        local direction = (targetRoot.Position - myRoot.Position).Unit
        targetRoot.AssemblyLinearVelocity = direction * 99999
        targetRoot.AssemblyAngularVelocity = Vector3.new(0, 9999, 0)
        
        task.wait(1)
        
        -- Grab gun
        return GrabGun()
    end
    
    return false
end

local function SendChatMessage(Message)
    local TextChatService = game:GetService("TextChatService")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    
    if TextChatService and TextChatService:FindFirstChild("ChatInputBarConfiguration") then
        local TextChannels = TextChatService:FindFirstChild("TextChannels")
        if TextChannels and TextChannels:FindFirstChild("RBXGeneral") then
            TextChannels.RBXGeneral:SendAsync(Message)
            return true
        end
    end
    
    local ChatEvents = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
    if ChatEvents then
        ChatEvents.SayMessageRequest:FireServer(Message, "All")
        return true
    end
    
    return false
end

function ExposeRole(WeaponName, RoleName)
    for _, Player in ipairs(Players:GetPlayers()) do
        local Character, Backpack = Player.Character, Player.Backpack
        if (Character and Character:FindFirstChild(WeaponName)) or (Backpack and Backpack:FindFirstChild(WeaponName)) then
            SendChatMessage(RoleName .. ": " .. Player.Name)
            return true
        end
    end
    return false
end

function NotifyRole(WeaponName, RoleName, Icon)
    for _, Player in ipairs(Players:GetPlayers()) do
        local Character, Backpack = Player.Character, Player.Backpack
        if (Character and Character:FindFirstChild(WeaponName)) or (Backpack and Backpack:FindFirstChild(WeaponName)) then
            WindUI:Notify({
                Title = "ESP",
                Content = RoleName .. ": " .. Player.Name,
                Duration = 5,
                Icon = Icon or "skull"
            })
            return true
        end
    end
    return false
end

-- Teleport to Murderer
TeleportTab:Button({
    Title = "Teleport to Murderer",
    Desc = "Teleport to murderer location",
    Icon = "skull",
    Callback = function()
        local murderer = GetMurderer()
        if murderer and murderer.Character then
            local root = GetRoot(murderer)
            if root then
                local myRoot = GetRoot(LocalPlayer)
                if myRoot then
                    myRoot.CFrame = root.CFrame * CFrame.new(0, 3, 0)
                    WindUI:Notify({Title = "Teleport", Content = "Teleported to Murderer!", Duration = 2})
                    return
                end
            end
        end

    end
})

-- Teleport to Map
TeleportTab:Button({
    Title = "Teleport to Map",
    Desc = "Teleport to map spawn",
    Icon = "map",
    Callback = function()
        local map = nil
        for _, child in ipairs(Workspace:GetChildren()) do
            if child:FindFirstChild("CoinContainer") and child:FindFirstChild("Spawns") then
                map = child
                break
            end
        end
        
        if map then
            local spawns = map:FindFirstChild("Spawns")
            if spawns then
                local children = spawns:GetChildren()
                if #children > 0 then
                    local spawn = children[math.random(1, #children)]
                    local myRoot = GetRoot(LocalPlayer)
                    if myRoot then
                        myRoot.CFrame = CFrame.new(spawn.Position) * CFrame.new(0, 3, 0)
                        WindUI:Notify({Title = "Teleport", Content = "Teleported to map!", Duration = 2})
                        return
                    end
                end
            end
        end
        WindUI:Notify({Title = "Teleport", Content = "Map not found!", Duration = 2})
    end
})

-- Teleport to Player
local function GetPlayerList()
    local names = {"Select Player"}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            table.insert(names, plr.Name)
        end
    end
    return names
end

TeleportTab:Dropdown({
    Title = "Teleport to Player",
    Desc = "Select a player to teleport to",
    Values = GetPlayerList(),
    Value = "Select Player",
    Callback = function(value)
        if value == "Select Player" then return end
        
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr.Name == value and plr.Character then
                local root = GetRoot(plr)
                if root then
                    local myRoot = GetRoot(LocalPlayer)
                    if myRoot then
                        myRoot.CFrame = root.CFrame * CFrame.new(0, 3, 0)
                        WindUI:Notify({Title = "Teleport", Content = "Teleported to " .. value .. "!", Duration = 2})
                        return
                    end
                end
            end
        end
        WindUI:Notify({Title = "Teleport", Content = value .. " not found!", Duration = 2})
    end
})
